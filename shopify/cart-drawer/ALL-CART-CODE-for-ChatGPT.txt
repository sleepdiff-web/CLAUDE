PURGE LABS (purgelabs.shop) - CART DRAWER CODE
==============================================

CONTEXT FOR CHATGPT
- Shopify store, live theme "Dawn" (15.x). Store currency GBP.
- This is a custom cart drawer that copies the Metabolae "Your order" drawer, recoloured to the brand red #913939.
- It is made of 4 files in the theme:
    1. snippets/cart-drawer.liquid              (the drawer markup and all the logic; settings at the top)
    2. sections/cart-drawer.liquid              (only renders the snippet, so the drawer can be re-rendered by the Section Rendering API)
    3. assets/component-cart-drawer-custom.css  (all the styles)
    4. assets/cart-drawer-custom.js             (the buttons: remove, - / +, Upgrade, optional offer card and shipping protection)
- Theme setting Cart > Cart type = "Drawer". Dawn's own assets/cart.js is still loaded.
- Bundles come from the Kaching Bundles app: "Buy 1", "Buy 2 Get 1 FREE", "Buy 3 Get 2 FREE".
  The free packs arrive as their own cart line priced 0. The snippet folds those free lines into the paid line of the
  same product, so Buy 3 Get 2 shows as one line of 5 packs.
- Two bundle modes are supported:
    * "tier" mode: bundles are separate variants (variant title contains "Buy 1" / "Buy 2 Get 1" / "Buy 3 Get 2").
      - / + and Upgrade add the other variant (/cart/add.js) and remove the current line (/cart/update.js).
    * "qty" mode: one variant added with a quantity (how Purge's Kaching bundle seems to work).
      - / + jump between 1, 3 and 5 packs and Upgrade sets 5, all through /cart/update.js on the line key.
      Free lines with a different key are zeroed through the data-also attribute.
- The green "Best price per pack unlocked" box shows once a product reaches 5 packs (the biggest bundle in the pack list).
  Otherwise the "BEST VALUE" upgrade card shows.
- The JS posts to /cart/add.js and /cart/update.js with sections "cart-drawer,cart-icon-bubble" and swaps #CartDrawer
  with the returned section HTML.
- Not yet confirmed: the exact structure of Purge's cart (output of /cart.js) with each Kaching bundle in it.
  The Upgrade price assumes Kaching charges for 3 of the 5 packs in Buy 3 Get 2.

{%- liquid
  assign mcd_facts = mcd_facts_data | split: '@@'
  assign mcd_pack_rows = mcd_pack_data | split: '@@'

  assign mcd_max_packs = 1
  assign mcd_top_paid = 0
  for prow in mcd_pack_rows
    assign pf = prow | split: '|||'
    assign pn = pf[1] | strip | plus: 0
    if pn > mcd_max_packs
      assign mcd_max_packs = pn
      assign mcd_top_paid = pf[2] | strip | plus: 0
    endif
  endfor
  if mcd_top_paid < 1 or mcd_top_paid > mcd_max_packs
    assign mcd_top_paid = mcd_max_packs
  endif

  assign mcd_pk_csv = ''
  assign mcd_mt_csv = ''
  for item in cart.items
    assign packs = 1
    assign pm = 'n'
    assign vt = item.variant.title | downcase
    for prow in mcd_pack_rows
      assign pf = prow | split: '|||'
      assign pkey = pf[0] | strip | downcase
      if pkey != blank and vt contains pkey
        assign packs = pf[1] | strip | plus: 0
        assign pm = 'y'
      endif
    endfor
    if packs < 1
      assign packs = 1
    endif
    assign mcd_pk_csv = mcd_pk_csv | append: packs | append: ','
    assign mcd_mt_csv = mcd_mt_csv | append: pm | append: ','
  endfor
  assign mcd_pk = mcd_pk_csv | split: ','
  assign mcd_mt = mcd_mt_csv | split: ','

  assign mcd_lines = 0
  assign mcd_savings = 0
  assign mcd_sp_key = ''
  assign mcd_up_in_cart = false
  assign mcd_is_max = false
  for item in cart.items
    if mcd_sp_variant > 0 and item.variant_id == mcd_sp_variant
      assign mcd_sp_key = item.key
      continue
    endif
    assign mcd_lines = mcd_lines | plus: 1
    if mcd_up_variant > 0 and item.variant_id == mcd_up_variant
      assign mcd_up_in_cart = true
    endif
    assign cu = item.variant.compare_at_price | default: 0
    if cu < item.original_price
      assign cu = item.original_price
    endif
    assign cl = cu | times: item.quantity
    assign ls = cl | minus: item.final_line_price
    if ls > 0
      assign mcd_savings = mcd_savings | plus: ls
    endif
    if item.final_line_price > 0 and item.variant_id != mcd_up_variant and mcd_max_packs > 1
      assign tot = 0
      for other in cart.items
        if other.product_id == item.product_id
          assign op = mcd_pk[forloop.index0] | plus: 0
          assign tot = op | times: other.quantity | plus: tot
        endif
      endfor
      if tot >= mcd_max_packs
        assign mcd_is_max = true
      endif
    endif
  endfor
  assign mcd_is_empty = false
  if mcd_lines == 0
    assign mcd_is_empty = true
  endif
  assign mcd_show_up = false
  if mcd_up_variant > 0 and mcd_lines > 0 and mcd_up_in_cart == false
    assign mcd_show_up = true
  endif
  assign mcd_show_sp = false
  if mcd_sp_variant > 0 and mcd_lines > 0
    assign mcd_show_sp = true
  endif
